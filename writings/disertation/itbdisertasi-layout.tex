% Juknis: nomor halaman bagian persiapan di tengah bawah; diagram menunjukkan
% jarak vertikal ~1{,}5\,cm dari tepi bawah kertas ke teks nomor. |\footskip|
% mengatur posisi relatif footer; nilai ini disetel agar mendekati diagram
% (verifikasi visual dengan |juknis_disertasi_Mei2019.pdf|).
\newcommand{\ITBGeomFootSkip}{42.5pt}

\newcommand{\ITBApplyPageGeometry}{%
  \geometry{%
    \ITBGeomPaper,%
    inner=\ITBGeomInner,%
    outer=\ITBGeomOuter,%
    top=\ITBGeomTop,%
    bottom=\ITBGeomBottom,%
    headheight=\ITBGeomHeadHeight,%
    headsep=\ITBGeomHeadSep,%
    footskip=\ITBGeomFootSkip,%
    twoside%
  }%
}
